{{- /* Ahrefs DR history sparkline for a directory, built from data/dr_history.yaml via the generic history-chart.html partial. Renders nothing until that directory has 2+ months of history. Call with a directory Page. */ -}} {{- $history := index hugo.Data.dr_history .File.ContentBaseName | default slice -}} {{- $n := len $history -}} {{- if ge $n 2 -}} {{- $values := slice -}} {{- range $history }}{{ $values = $values | append (float .dr) }}{{ end -}} {{- $first := index $history 0 -}} {{- $last := index $history (sub $n 1) -}} {{- $delta := sub (float $last.dr) (float $first.dr) -}} {{- $month := (time.AsTime $first.date).Format "Jan 2006" -}} {{- $deltaText := printf "no change since %s" $month -}} {{- if gt $delta 0.0 -}} {{- $deltaText = printf "+%.0f since %s" $delta $month -}} {{- else if lt $delta 0.0 -}} {{- $deltaText = printf "%.0f since %s" $delta $month -}} {{- end -}} {{- $ariaLabel := printf "Ahrefs DR trend: %v in %s, now %v" $first.dr $month $last.dr -}} {{ partial "history-chart.html" (dict "values" $values "id" (printf "dr-chart-fade-%s" .File.ContentBaseName) "deltaText" $deltaText "ariaLabel" $ariaLabel ) }} {{- end -}}
